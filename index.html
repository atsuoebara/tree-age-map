/* ========================= 実走線 同じ場所の重複描画を抑える ========================= */ function drawRoutes(selectedActivities) { /* * 約10m単位で座標を丸め、 * 同じ区間を何度走っても * 1回だけ描画する */ const drawnSegments = new Set(); const roundCoord = value => Math.round(value * 10000) / 10000; selectedActivities.forEach(activity => { const latlngs = activity.latlngs; for (let i = 1; i < latlngs.length; i++) { const a = latlngs[i - 1]; const b = latlngs[i]; const aKey = roundCoord(a[0]) + ',' + roundCoord(a[1]); const bKey = roundCoord(b[0]) + ',' + roundCoord(b[1]); /* * A→B と B→A を同じ道路として扱う */ const segmentKey = aKey < bKey ? aKey + '|' + bKey : bKey + '|' + aKey; if (drawnSegments.has(segmentKey)) { continue; } drawnSegments.add(segmentKey); const route = L.polyline( [a, b], { color: ROUTE_COLOR, weight: 1.5, opacity: 0.30, lineCap: 'round', lineJoin: 'round' } ); routeGroup.addLayer(route); } }); }
